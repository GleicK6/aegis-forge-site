# AEGIS — politique de crawl
# Site statique vitrine. Indexation autorisée par les moteurs légitimes.
#
# Doctrine : la citation est ouverte, l'entraînement reste fermé.
# Depuis février 2026, OpenAI et Anthropic exposent des user-agents distincts
# pour l'entraînement, l'indexation de recherche et le fetch déclenché par un
# utilisateur. On distingue les trois au lieu de tout bloquer :
#   - recherche + fetch utilisateur : autorisés sur tout le site (visibilité)
#   - entraînement de modèles       : restreint à la page Octane (acquisition)
#   - collecteurs opaques / SEO     : bloqués
# Côté aegis-resilience.com, l'allowlist Octane est sans effet (page absente).

User-agent: *
Allow: /
Disallow: /rss/.orig$
Disallow: /rss/*.orig

# ── Recherche IA & fetch utilisateur — accès complet ──────────────
# Ces agents alimentent les réponses citées (ChatGPT Search, Claude,
# Perplexity). Les bloquer revient à se rendre invisible sur le canal
# où les collectivités formulent leurs recherches.

User-agent: OAI-SearchBot
Allow: /

User-agent: ChatGPT-User
Allow: /

User-agent: Claude-SearchBot
Allow: /

User-agent: Claude-User
Allow: /

User-agent: PerplexityBot
Allow: /

# ── Entraînement de modèles — allowlist Octane uniquement ─────────
# La règle Allow plus spécifique (path complet) prime sur le Disallow racine
# pour les agents qui respectent la spec étendue Google/OpenAI.
# Octane a besoin d'une visibilité IA pour l'acquisition ; le travail
# éditorial des autres pages reste hors des corpus d'entraînement.

User-agent: GPTBot
Allow: /forge-octane.html
Allow: /llms.txt
Allow: /octane-og.png
Disallow: /

User-agent: ClaudeBot
Allow: /forge-octane.html
Allow: /llms.txt
Disallow: /

# Noms hérités des anciennes documentations Anthropic — conservés par
# précaution tant que d'anciens clients les émettent encore.

Disallow: /

User-agent: Claude-Web
Allow: /forge-octane.html

Disallow: /

User-agent: meta-externalagent
Allow: /forge-octane.html

# Google-Extended couvre Gemini et le grounding Vertex AI. Google ne sépare
# pas entraînement et citation sur cet agent : l'ouvrir donnerait des
# citations Gemini au prix de l'entraînement. Choix actuel = restreint à
# Octane, cohérent avec la ligne ci-dessus. Passer en "Allow: /" si la
# visibilité Gemini devient prioritaire.
User-agent: Google-Extended
